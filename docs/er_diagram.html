Mimari: Schema-per-Tenant
Şema: PostgreSQL
Tablo Sayısı: 12 Tablo

    %% Master Schema (public)
    TENANTS {
        bigint id PK
        varchar tenant_id UK "Kiracı Kodu"
        varchar schema_name UK "PostgreSQL Şema Adı"
        varchar company_name "İşletme Ünvanı"
        boolean is_active "Aktiflik"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    %% Tenant Schema - Stok & Ürün Modülü
    CATEGORIES {
        bigint id PK
        varchar code UK "Kategori Kodu"
        varchar name "Kategori Adı"
        text description
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    PRODUCTS {
        bigint id PK
        bigint category_id FK "Kategori Referansı"
        varchar code UK "Ürün Kodu"
        varchar name "Ürün Adı"
        varchar base_unit "Temel Birim"
        numeric tax_rate "KDV Oranı"
        jsonb attributes "Dinamik Özellikler"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    PRODUCT_VARIANTS {
        bigint id PK
        bigint product_id FK "Ana Ürün"
        varchar sku UK "Stok Kodu (SKU)"
        varchar barcode "Barkod"
        varchar variant_name "Varyant Tanımı"
        numeric purchase_price "Alış Fiyatı"
        numeric sale_price "Satış Fiyatı"
        integer stock_quantity "Fiili Depo Stoğu"
        integer reserved_stock "Rezerve Sipariş Stoğu"
        jsonb attributes "Renk, Beden vb."
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    %% Tenant Schema - Cari Hesap Modülü
    BUSINESS_PARTNERS {
        bigint id PK
        varchar partner_type "CUSTOMER / SUPPLIER"
        varchar name "Cari Adı"
        varchar company_title "Resmi Ünvan"
        varchar tax_number "VKN"
        varchar email "E-Posta"
        varchar phone "Telefon"
        text address "Adres"
        jsonb metadata "Ek Bilgiler"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    %% Tenant Schema - Teklif Modülü
    QUOTATIONS {
        bigint id PK
        varchar quotation_number UK "Teklif No"
        varchar type "PURCHASE / SALES"
        bigint partner_id FK "Cari Hesap"
        varchar status "DRAFT / SENT / ACCEPTED"
        timestamp issue_date
        timestamp valid_until
        varchar currency
        numeric total_amount "Genel Toplam"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    QUOTATION_ITEMS {
        bigint id PK
        bigint quotation_id FK "Teklif"
        bigint variant_id FK "Ürün Varyantı"
        varchar description
        integer quantity "Miktar"
        numeric unit_price "Birim Fiyat"
        numeric subtotal "Ara Toplam"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    %% Tenant Schema - Sipariş Modülü
    ORDERS {
        bigint id PK
        varchar order_number UK "Sipariş No"
        varchar order_type "SALES_ORDER / PURCHASE_ORDER"
        bigint partner_id FK "Cari Hesap"
        bigint quotation_id FK "İlişkili Teklif"
        varchar status "DRAFT / CONFIRMED / COMPLETED"
        timestamp order_date
        timestamp delivery_date
        numeric total_amount "Genel Toplam"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    ORDER_ITEMS {
        bigint id PK
        bigint order_id FK "Sipariş"
        bigint variant_id FK "Ürün Varyantı"
        varchar description
        integer quantity "Miktar"
        numeric unit_price "Birim Fiyat"
        numeric subtotal "Ara Toplam"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    %% Tenant Schema - İrsaliye Modülü
    WAYBILLS {
        bigint id PK
        varchar waybill_number UK "İrsaliye No"
        varchar type "DISPATCH / RECEIPT"
        bigint partner_id FK "Cari Hesap"
        bigint order_id FK "İlişkili Sipariş"
        varchar status "DRAFT / DISPATCHED / DELIVERED"
        timestamp dispatch_date
        varchar carrier_company "Taşıyıcı"
        varchar tracking_number "Takip Kodu"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    WAYBILL_ITEMS {
        bigint id PK
        bigint waybill_id FK "İrsaliye"
        bigint variant_id FK "Ürün Varyantı"
        varchar description
        integer quantity "Sevk Miktarı"
        numeric unit_price "Birim Fiyat"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    %% Tenant Schema - Kullanıcılar & Denetim
    USERS {
        bigint id PK
        varchar username UK "Kullanıcı Adı"
        varchar email UK "E-Posta"
        varchar password_hash "Şifre Özeti"
        varchar full_name "Ad Soyad"
        varchar role "Rol"
        boolean is_active "Aktif"
        timestamp created_at
        timestamp updated_at
        bigint version
    }

    AUDIT_LOGS {
        bigint id PK
        varchar action "CREATE / UPDATE / DELETE"
        varchar entity_type "Varlık Tipi"
        bigint entity_id "Varlık ID"
        varchar performed_by "Kullanıcı"
        jsonb details "İşlem Detayı"
        timestamp performed_at
    }

    %% İlişkiler
    CATEGORIES ||--o{ PRODUCTS : "içerir"

    BUSINESS_PARTNERS ||--o{ QUOTATIONS : "teklif"
    BUSINESS_PARTNERS ||--o{ ORDERS : "sipariş"
    BUSINESS_PARTNERS ||--o{ WAYBILLS : "irsaliye"

    QUOTATIONS ||--|{ QUOTATION_ITEMS : "satırlar"
    PRODUCT_VARIANTS ||--o{ QUOTATION_ITEMS : "kalem"

    QUOTATIONS |o--o{ ORDERS : "dönüştürülür"
    ORDERS ||--|{ ORDER_ITEMS : "satırlar"
    PRODUCT_VARIANTS ||--o{ ORDER_ITEMS : "kalem"

    ORDERS |o--o{ WAYBILLS : "irsaliyeye dönüştürülür"
    WAYBILLS ||--|{ WAYBILL_ITEMS : "satırlar"
    PRODUCT_VARIANTS ||--o{ WAYBILL_ITEMS : "sevk edilir"

🏬 Çoklu Kiracılık (Multi-Tenancy)

Master şemada yer alan tenants tablosu ile PostgreSQL üzerinde her işletmeye özel izole şema (schema_name) yönetimi sağlanır.

tenants

📦 Stok & Ürün Yönetimi

Kategoriler, genel ürün şablonu (Product) ve fiziksel barkod/SKU/fiyat/stok hareketlerini tutan dinamik JSONB destekli varyantlar (ProductVariant).

categories products product_variants

🤝 Cari Hesaplar (CRM)

Müşteri (Customer), Tedarikçi (Supplier) veya her ikisi olan ticari iş ortakları, vergi bilgileri ve dinamik metadata alanları.

business_partners

📑 B2B Belge Akışı (Order-to-Cash)

Teklif (Quotation) onaylandığında Siparişe (Order), sipariş hazırlandığında ise fiziksel sevkiyat için İrsaliyeye (Waybill) dönüştürülür.

quotations orders waybills

🔐 Güvenlik ve Denetim (Audit)

Kiracı bazlı kullanıcı kimlik doğrulaması ve her varlık üzerinde gerçekleştirilen CRUD/durum değişikliklerinin JSON diff ile saklandığı loglama.

users audit_logs
